// arrays are the collection of data, they are used to store multiple values in a single variable

let fruit = ["apple", "banana", "orange", "grape", "kiwi"];
fruit.push("mango"); // adds "mango" to the end of the array
fruit.unshift("strawberry"); // adds "strawberry" to the beginning of the array


fruit.pop(); // removes the last element from the array
fruit.shift(); // removes the first element from the array
console.log(fruit); // ["apple", "banana", "orange", "grape", "kiwi", "mango"]


// write notes for these all methods 
// push() - adds one or more elements to the end of an array
// unshift() - adds one or more elements to the beginning of an array
// pop() - removes the last element from an array and returns that element
// shift() - removes the first element from an array and returns that element

// filter method

let ages = [12, 17, 25, 30, 15, 18];

let adultts = ages.filter(function(age){
    return age >= 18; // Filter out ages less than 18
})


console.log(adultts); // Output: [25, 30, 18]
console.log(ages); // Output: [12, 17, 25, 30, 15, 18] (original array remains unchanged)




// reduce method

let marks = [10, 30, 50];

let total = marks.reduce (function(sum , mark){
    return sum + mark; // Calculate the total sum of marks
} , 10); // Initial value of sum is set to 10
console.log(total); // Output: 90

// promises (pending , resolve , reject)

function loginUser(username, password) {
    if (username === "admin" && password === "password") {
        console.log("Login successful");
    } else {
        console.log("Login failed");
    }

}

loginUser("admin", "password");



// same code in promise (resolve , reject)

function loginUser(username, password) {
    return new Promise((resolve, reject) => {
        if (username === "admin" && password === "password") {
            resolve("Login successful");
        } else {
            reject("Login failed");
        }
    });

}

loginUser("admin", "password").then((result) => {
    console.log(result);
}).catch((err) => {
    console.error(err);
});
