let fruit = ["apple", "banana", "orange", "grape", "kiwi"];
fruit.push("mango"); // adds "mango" to the end of the array
fruit.unshift("strawberry"); // adds "strawberry" to the beginning of the array


fruit.pop(); // removes the last element from the array
fruit.shift(); // removes the first element from the array
console.log(fruit); // ["apple", "banana", "orange", "grape", "kiwi", "mango"]


// write notes for these all methods 
// push() - adds one or more elements to the end of an array
// unshift() - adds one or more elements to the beginning of an array
// pop() - removes the last element from an array and returns that element
// shift() - removes the first element from an array and returns that element

// filter method

let ages = [12, 17, 25, 30, 15, 18];

let adultts = ages.filter(function(age){
    return age >= 18; // Filter out ages less than 18
})


console.log(adultts); // Output: [25, 30, 18]
console.log(ages); // Output: [12, 17, 25, 30, 15, 18] (original array remains unchanged)




// reduce method

let marks = [10, 30, 50];

let total = marks.reduce (function(sum , mark){
    return sum + mark; // Calculate the total sum of marks
} , 10); // Initial value of sum is set to 10

// promises (pending , resolve , reject)

function loginUser(username, password) {
    if (username === "admin" && password === "password") {
        console.log("Login successful");
    } else {
        console.log("Login failed");
    }

}

loginUser("admin", "password");



// same code in promise (resolve , reject)

function loginUser(username, password) {
    return new Promise((resolve, reject) => {
        if (username === "admin" && password === "password") {
            resolve("Login successful");
        } else {
            reject("Login failed");
        }
    });

}

loginUser("admin", "password").then((result) => {
    console.log(result);
}).catch((err) => {
    console.error(err);
});

// Important Example

function getUsers() {
    return new Promise((resolve) => {
        setTimeout(() => {
            resolve("User loaded");
        }, 2000);
    });
}

async function loadUsers() {
    let result = await getUsers();
    console.log(result);
}


console.log("Start");
loadUsers();
console.log("Finish");

// explanation of async and await function. 
// in this example, the loadUsers function is an async function that calls the getUsers function, which returns a promise. The await keyword is used to wait for the promise to resolve before continuing with the execution of the code. This allows for asynchronous code to be written in a more synchronous manner, making it easier to read and understand.

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
</head>
<body>
    <h1>My Fetch API Example</h1>
    <button id="fetchButton">Fetch Data</button>
    <div id="output"></div>

    <script>
        const fetchButton = document.querySelector('#fetchButton');
        const output = document.querySelector('#output');

        fetchButton.addEventListener('click', () => {
            fetch('https://jsonplaceholder.typicode.com/users')
                .then(response => {
                    if (!response.ok) {
                        throw new Error('Network response was not ok');
                    }
                    return response.json();
                })
                .then(data => {
                    output.innerHTML = ` <ul>
            ${data.map(user => `<li>${user.name} - ${user.website} - ${user.phone}</li>`).join('')}
        </ul>`;
                })
                .catch(error => {
                    console.error('There was a problem with the fetch operation:', error);
                });
        });
    </script>
</body>
</html>
